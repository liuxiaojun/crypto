```
import backtrader as bt

class DoubleMaStrategy(bt.Strategy):
    """ 双均线金叉/死叉策略 """
    # 1. 定义参数区： 将策略参数暴漏出来，便于后续做参数寻优
    params = (('fast',10),('slow',30))

    def __init__(self):
        # 2. 初始化区： 声明指标与计算图
        self.ma_fast = bt.indicators.SMA(self.data.close,period=self.fast)
        self.ma_slow = bt.indicators.SMA(self.data.close,period=self.p.slow)

        # CrossOver指标： 快线上穿慢线返回1，下穿返回-1，无交叉返回0
        self.crossover = bt.indicators.CrossOver(self.ma_fast,self.ma_slow)

    def next(self):
        # 3. 步进逻辑区：按时间序列逐跟K线执行的交易逻辑
        if not self.position: # 状态机：如果当前账户处于空仓状态
            if self.crossover > 0: # 捕捉到金叉信号
                self.buy()
        elif self.crossover < 0: # 状态机： 如果当前持仓，且捕捉到死叉信号
            self.close()
```

bt.indicators.SMA 调用backtrader 内置的简单移动平均线指标    
self.data.close   调用当前数据源的收盘价作为计算输入    
period=self.fast  设置计算周期，值来自策略参数self.p.fast (通常定义为5/10/20等短期周期)    
self.ma_fast      将指标实例保存为策略实例的属性，方便在next() 中引用
